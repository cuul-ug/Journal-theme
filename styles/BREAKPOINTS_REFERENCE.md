/**
 * RESPONSIVE MEDIA QUERIES BREAKPOINTS
 * African Journal of Library and Information Science Theme
 *
 * Mobile-first approach: Start with mobile styles, enhance with media queries
 */

// ============================================
// STANDARD BREAKPOINT DEFINITIONS
// ============================================

// Extra Small (xs) - Mobile devices
// Default/Mobile-first approach
// 320px - 575px

// Small (sm) - Large phones to small tablets
// @media (min-width: 576px) { }

// Medium (md) - Tablets
// @media (min-width: 768px) { }

// Large (lg) - Small laptops
// @media (min-width: 992px) { }

// Extra Large (xl) - Desktops
// @media (min-width: 1200px) { }

// 2XL - Large monitors
// @media (min-width: 1400px) { }

// ============================================
// DEVICE-SPECIFIC MEDIA QUERIES
// ============================================

// Mobile devices (landscape orientation)
// @media (orientation: landscape) and (max-height: 600px) { }

// Tablet devices (portrait)
// @media only screen and (min-device-width: 768px) and (max-device-width: 1024px) and (orientation: portrait) { }

// Tablet devices (landscape)
// @media only screen and (min-device-width: 768px) and (max-device-width: 1024px) and (orientation: landscape) { }

// High pixel density (Retina displays)
// @media (-webkit-min-device-pixel-ratio: 2), (min-resolution: 192dpi) { }

// ============================================
// FEATURE-BASED MEDIA QUERIES
// ============================================

// Touch devices
// @media (hover: none) and (pointer: coarse) { }

// Non-touch devices (hover available)
// @media (hover: hover) and (pointer: fine) { }

// Dark mode preference
// @media (prefers-color-scheme: dark) { }

// Light mode preference (default)
// @media (prefers-color-scheme: light) { }

// Reduced motion preference (accessibility)
// @media (prefers-reduced-motion: reduce) { }

// High contrast preference (accessibility)
// @media (prefers-contrast: more) { }

// Reduced transparency (accessibility)
// @media (prefers-reduced-transparency: reduce) { }

// ============================================
// EMERGING SCREEN TECHNOLOGY QUERIES
// ============================================

// Foldable/multi-screen devices
// @media (fold-left: 0px) { }
// @media (fold-right: 100vw) { }

// Landscape orientation queries
// @media (orientation: landscape) { }

// Landscape mobile (wide and short)
// @media (orientation: landscape) and (max-height: 500px) { }

// ============================================
// VIEWPORT HEIGHT-BASED QUERIES
// ============================================

// Very tall screens
// @media (min-height: 1080px) { }

// Short screens (limited vertical space)
// @media (max-height: 600px) { }

// ============================================
// PRINT MEDIA QUERIES
// ============================================

// Print styles
// @media print { }

// Print specific breakpoints
// @media print and (min-width: 1200px) { }
